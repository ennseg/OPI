# 1. compile — компиляция
mvn compile

# 2. build — сборка в WAR+JAR
mvn package -DskipTests

# 3. clean — очистка
mvn clean

# 4. test — запуск тестов (включает build)
mvn test

# 5. xml — валидация XML
mvn xml:validate

# 6. music — сборка с музыкой
mvn package -P music -DskipTests

# 7. native2ascii — конвертация локализации
mvn generate-resources -P native2ascii

# 8. alt — альтернативная версия
mvn package -P alt -DskipTests

# 9. team — 3 ревизии из git
mvn package -P team -DskipTests

# 10. history — поиск последней рабочей ревизии
mvn validate -P history
